# ========================================================
# IC10 - POWER MONITOR VOOR MODULAR CONSOLES
# Batch-uitlezing van accubank, watt-berekening en EMA
# ========================================================

# Pin Toewijzingen
alias MemTotalCharge d0
alias MemDeltaWatt   d1
alias MemAvgWatt     d2

# Constanten & Hash
define BATTERY_HASH     HASH("StructureBatteryLarge")
define TICK_DELTA_SEC   0.5     # Duur van 1 Stationeers tick (sec)
define ALPHA            0.1     # Dempingsfactor rollend gemiddelde

# Register Aliases
alias CurrentCharge  r0
alias PrevCharge     r1
alias DeltaJoules    r2
alias InstantWatt    r3
alias AvgWatt        r4
alias Initialized    r5
alias TempReg        r6

main:
    yield

    # 1. Lees totale opgeslagen energie van alle batterijen via batch
    lb CurrentCharge BATTERY_HASH Charge 1

    # 2. Eerste tick initialiseren om pieken te voorkomen
    bnez Initialized calc_power
    move PrevCharge CurrentCharge
    move AvgWatt 0
    move Initialized 1
    j store_data

calc_power:
    # 3. Bereken actuele wattages (Watt = Delta Joules / Delta Tijd)
    sub DeltaJoules CurrentCharge PrevCharge
    div InstantWatt DeltaJoules TICK_DELTA_SEC

    # 4. Rollend exponentieel gemiddelde (EMA) bijwerken
    sub TempReg InstantWatt AvgWatt
    mul TempReg TempReg ALPHA
    add AvgWatt AvgWatt TempReg

    # Werk PrevCharge bij voor de volgende cyclus
    move PrevCharge CurrentCharge

store_data:
    # 5. Waardes wegschrijven naar Logic Memories voor consoles
    s MemTotalCharge Setting CurrentCharge
    s MemDeltaWatt Setting InstantWatt
    s MemAvgWatt Setting AvgWatt

    j main
Uitleg & Afstelling: Dit script leest via batching (lb) de totale lading van alle grote batterijen tegelijk uit en berekent het netto laad- of ontlaadvermogen per seconde. Met de parameter ALPHA (standaard 0.1) pas je de reactiesnelheid van het gemiddelde aan; verlaag naar 0.05 voor een vlakkere curve op de Modular Consoles schermen.

# ========================================================
# IC10 - FILTRATIE & LEIDING TELEMETRIE CONTROLLER
# Procesregeling, filterbewaking en batch-alarmering
# ========================================================

# Pin Toewijzingen
alias Filtration    d0
alias StorageTank   d1
alias PipeIn        d2
alias PipeOut       d3
alias MemTank       d4
alias MemFilter     d5

# Grenswaarden & Hash
define MIN_IN_PRESS       20.0    # kPa: Minimum toevoerdruk voor inschakelen
define MAX_PIPE_PRESS     50000.0 # kPa: Maximale veilige leidingdruk (50 MPa)
define MAX_TANK_PRESS     55000.0 # kPa: Maximale tankdoeldruk (55 MPa)
define MIN_SAFE_TEMP      200.0   # Kelvin: Ondergrens leidingtemperatuur
define MAX_SAFE_TEMP      350.0   # Kelvin: Bovengrens leidingtemperatuur (~77 C)
define MIN_FILTER_PCT     1.0     # %: Minimum filterstatus voor noodstop
define ALARM_LIGHT_HASH   HASH("StructureFlashingLight")

# Register Aliases
alias InPress       r0
alias OutPress      r1
alias TankPress     r2
alias OutTemp       r3
alias Filter0       r4
alias Filter1       r5
alias MinFilter     r6
alias FaultFlag     r7
alias RunState      r8

main:
    yield

    # 1. Uitlezen sensordata en tankdruk
    l InPress   PipeIn Pressure
    l OutPress  PipeOut Pressure
    l TankPress StorageTank Pressure
    l OutTemp   PipeOut Temperature

    # 2. Slijtage van beide filters uitlezen (Slot 0 en 1)
    ls Filter0 Filtration 0 Quantity
    ls Filter1 Filtration 1 Quantity
    min MinFilter Filter0 Filter1

    # 3. Integriteit van leidingen en tank controleren
    sgt r9 OutPress MAX_PIPE_PRESS
    sgt r10 TankPress MAX_TANK_PRESS
    or FaultFlag r9 r10

    slt r9 OutTemp MIN_SAFE_TEMP
    sgt r10 OutTemp MAX_SAFE_TEMP
    or r9 r9 r10
    or FaultFlag FaultFlag r9

    # 4. Filterprocescondities bepalen
    sgt r9 InPress MIN_IN_PRESS
    slt r10 TankPress MAX_TANK_PRESS
    and RunState r9 r10

    sgt r9 MinFilter MIN_FILTER_PCT
    and RunState RunState r9

    # Direct uitschakelen bij leidingfout of overdruk
    select RunState FaultFlag 0 RunState
    s Filtration On RunState

    # 5. Visueel alarm triggeren via batch
    sb ALARM_LIGHT_HASH On FaultFlag

    # 6. Schrijf data weg naar Logic Memories
    s MemTank Setting TankPress
    s MemFilter Setting MinFilter

    j main
Uitleg & Afstelling: Schakelt de filtratie-unit automatisch uit als de tank vol is of zodra een van de interne filters onder 1% zakt via ls Quantity. Pas MAX_PIPE_PRESS aan naar 60000 bij Heavy-Duty leidingen en stel MIN_SAFE_TEMP af op het kookpunt van het specifieke gas om condensatie in de leidingen te weren.

# ========================================================
# IC10 - FILTRATIEKAMER HVAC & NOODAFZUIGING
# Bestuurt 4x Powered Vent Large (2 extractie, 2 toevoer)
# Drukbalans en geforceerde noodontluchting bij calamiteiten
# ========================================================

# Pin Toewijzingen
alias VentExt1     d0
alias VentExt2     d1
alias VentSup1     d2
alias VentSup2     d3
alias SensorFilt   d4
alias MemHazard    d5

# Grenswaarden & Constanten
define PRESS_MIN        100.0   # kPa: Minimale kamerdoeldruk
define PRESS_MAX        102.0   # kPa: Maximale kamerdoeldruk
define MAX_SAFE_T       323.15  # Kelvin: Max veilige temperatuur (50 C)
define MAX_TOXIC_RATIO  0.001   # Ratio: Max 0.1% toxisch gas (CH4, X, N2O, H2O)
define MODE_INWARD      0       # Mode 0: Afzuigen naar leidingsysteem
define MODE_OUTWARD     1       # Mode 1: Blazen in de kamer

# Register Aliases
alias FiltPress    r0
alias FiltTemp     r1
alias ToxicSum     r2
alias HazardActive r3
alias SupOn        r4
alias ExtOn        r5
alias TempReg      r6

main:
    yield

    # 1. Atmosfeer en master gevarenbus inlezen
    l FiltPress   SensorFilt Pressure
    l FiltTemp    SensorFilt Temperature
    l HazardActive MemHazard Setting

    # 2. Multi-gas toxiciteit optellen (Methane, Pollutant, N2O, Water)
    l ToxicSum  SensorFilt RatioPollutant
    l TempReg   SensorFilt RatioNitrousOxide
    add ToxicSum ToxicSum TempReg
    l TempReg   SensorFilt RatioMethane
    add ToxicSum ToxicSum TempReg
    l TempReg   SensorFilt RatioWater
    add ToxicSum ToxicSum TempReg

    # 3. Lokale gevarencontrole (temperatuur of toxisch lek)
    sgt TempReg ToxicSum MAX_TOXIC_RATIO
    or HazardActive HazardActive TempReg
    sgt TempReg FiltTemp MAX_SAFE_T
    or HazardActive HazardActive TempReg

    # 4. Bepaal status van ventilatoren
    beqz HazardActive normal_hvac

    # ALARM: Toevoer per direct afsluiten, maximale extractie aan
    move SupOn 0
    move ExtOn 1
    j apply_hvac

normal_hvac:
    # NORMALE REGELING: Handhaaf druk tussen PRESS_MIN en PRESS_MAX
    slt SupOn FiltPress PRESS_MIN
    sgt ExtOn FiltPress PRESS_MAX

apply_hvac:
    # 5. Extractieventilatoren instellen en schakelen
    s VentExt1 Mode MODE_INWARD
    s VentExt2 Mode MODE_INWARD
    s VentExt1 On ExtOn
    s VentExt2 On ExtOn

    # 6. Toevoerventilatoren instellen en schakelen
    s VentSup1 Mode MODE_OUTWARD
    s VentSup2 Mode MODE_OUTWARD
    s VentSup1 On SupOn
    s VentSup2 On SupOn

    j main
Uitleg & Afstelling: Balanceert de kamerdruk strak tussen 100 en 102 kPa met de 4 grote ventilatoren. Bij detectie van brandbare of giftige gassen (Methane, X, N2O, stoom) of hitte slaat de toevoer onmiddellijk dicht en zuigen beide extractieventilatoren de ruimte op maximale capaciteit vacuüm.

# ========================================================
# IC10 - SLUIS 2 & FILTRATIE CONTAINMENT CONTROLLER
# Stuurt Airlock Door 2A & Blast Door 2B aan
# Bevat noodevacuatie via de Occupancy Sensor & MemHazard bus
# ========================================================

# Pin Toewijzingen
alias Door2A       d0
alias Door2B       d1
alias SensorFilt   d2
alias SensorAir2   d3
alias OccFilt      d4
alias MemHazard    d5

# Grenswaarden Filtratiekamer & Veiligheid
define MIN_SAFE_P       80.0     # kPa: Minimale veilige kamerdruk
define MAX_SAFE_P       125.0    # kPa: Maximale veilige kamerdruk
define MAX_SAFE_T       323.15   # Kelvin: Maximale veilige temperatuur (50 C)
define MAX_TOXIC_RATIO  0.001    # Ratio: Max 0.1% toxisch (CH4, X, N2O, H2O)
define MAX_CO2_RATIO    0.02     # Ratio: Max 2.0% CO2

# Register Aliases
alias FiltPress    r0
alias FiltTemp     r1
alias ToxicSum     r2
alias HazardFlag   r3
alias D2AOpen      r4
alias D2BOpen      r5
alias PlayerInFilt r6
alias TempReg      r7

main:
    yield

    # 1. Sensordata en bezetting in de filtratiekamer uitlezen
    l FiltPress    SensorFilt Pressure
    l FiltTemp     SensorFilt Temperature
    l PlayerInFilt OccFilt Quantity

    # 2. Multi-gas toxiciteit optellen (Methane, Pollutant, N2O, Water)
    l ToxicSum SensorFilt RatioPollutant
    l TempReg  SensorFilt RatioNitrousOxide
    add ToxicSum ToxicSum TempReg
    l TempReg  SensorFilt RatioMethane
    add ToxicSum ToxicSum TempReg
    l TempReg  SensorFilt RatioWater
    add ToxicSum ToxicSum TempReg

    # 3. Integrale gevarendiagnose stellen
    slt HazardFlag FiltPress MIN_SAFE_P
    sgt TempReg FiltPress MAX_SAFE_P
    or HazardFlag HazardFlag TempReg

    sgt TempReg FiltTemp MAX_SAFE_T
    or HazardFlag HazardFlag TempReg

    sgt TempReg ToxicSum MAX_TOXIC_RATIO
    or HazardFlag HazardFlag TempReg

    l TempReg SensorFilt RatioCarbonDioxide
    sgt TempReg TempReg MAX_CO2_RATIO
    or HazardFlag HazardFlag TempReg

    # Controleer of Sluis 2 zelf besmet is
    l TempReg SensorAir2 RatioPollutant
    sgt TempReg TempReg MAX_TOXIC_RATIO
    or HazardFlag HazardFlag TempReg

    # Deurstatussen inlezen (0 = Dicht, 1 = Open)
    l D2AOpen Door2A Open
    l D2BOpen Door2B Open

    # 4. Deurbeveiliging & Interlock
    beqz HazardFlag normal_ops

    # ALARM: Controleer of er een speler geëvacueerd moet worden
    bgtz PlayerInFilt evacuate_player

    # Geen speler binnen: Vergrendel beide deuren hermetisch
    s Door2B Lock 1
    s Door2B Open 0
    s Door2A Lock 1
    s Door2A Open 0
    j update_hazard_bus

evacuate_player:
    # Ontgrendel Blast Door 2B voor vluchtroute, buitendeur potdicht houden
    s Door2A Lock 1
    s Door2A Open 0
    s Door2B Lock 0
    j update_hazard_bus

normal_ops:
    # Standaard interlock over de 2-blokken buffer
    s Door2A Lock D2BOpen
    s Door2B Lock D2AOpen

update_hazard_bus:
    # 5. Schrijf status weg naar shared gevarenbus
    s MemHazard Setting HazardFlag

    j main
Uitleg & Afstelling: Bewaakt de 2-blokken decontaminatiezone en de binnenste Blast Door 2B. Als er giftige gassen ontsnappen terwijl er een technicus in de filtratieruimte staat (OccFilt > 0), blijft Blast Door 2B ontgrendeld voor evacuatie naar de sluis, terwijl Deur 2A potdicht gaat om de basis te beschermen.

# ========================================================
# IC10 - SLUIS 1 HALLWAY ACCESS CONTROLLER
# Stuurt Hallway Blast Door 1A & Airlock Door 1B aan
# Sluit de centrale gang hermetisch af bij calamiteiten
# ========================================================

# Pin Toewijzingen
alias Door1A       d0
alias Door1B       d1
alias SensorAir1   d2
alias OccAir1      d3
alias MemHazard    d4
alias MemStatus1   d5

# Grenswaarden
define MAX_TOXIC_RATIO  0.001    # Maximaal toelaatbare toxische fractie

# Register Aliases
alias HazardActive r0
alias D1AOpen      r1
alias D1BOpen      r2
alias InTransit    r3
alias ToxicCheck   r4
alias StatusVal    r5
alias TempReg      r6

main:
    yield

    # 1. Lees gevarenbus en sensorwaarden in
    l HazardActive MemHazard Setting
    l InTransit    OccAir1 Quantity
    l D1AOpen      Door1A Open
    l D1BOpen      Door1B Open

    # 2. Controleer op binnengedragen gifgassen
    l ToxicCheck SensorAir1 RatioPollutant
    l TempReg    SensorAir1 RatioMethane
    add ToxicCheck ToxicCheck TempReg

    # 3. Beveiligingsafhandeling
    beqz HazardActive check_transit

    # BIJ GEVAAR IN HET COMPLEX:
    # Vergrendel Door 1B om de centrale hallway te beschermen
    s Door1B Lock 1
    s Door1B Open 0
    s Door1A Lock D1BOpen
    move StatusVal -1
    j write_status

check_transit:
    # 4. Normale sluisinterlock met bezettingsbewaking
    s Door1A Lock D1BOpen
    s Door1B Lock D1AOpen

    # Statuscode bepalen (-1: Alarm, 0: Standby, 1: 1A Open, 2: 1B Open, 3: Bezet)
    move StatusVal 0
    select StatusVal D1AOpen 1 StatusVal
    select StatusVal D1BOpen 2 StatusVal
    bgtz InTransit mark_occupied
    j write_status

mark_occupied:
    move StatusVal 3

write_status:
    # 5. Status exporteren naar Logic Memory
    s MemStatus1 Setting StatusVal

    j main
Uitleg & Afstelling: Bestuurt de primaire toegang vanaf de basis via Blast Door 1A. Zodra de gevarenbus MemHazard actief wordt, gaat binnendeur 1B onmiddellijk op slot, waardoor besmetting vanuit de tussengang nooit de centrale hallway kan bereiken.

    l ToxicSum  SensorRoom RatioPollutant
    l TempReg   SensorRoom RatioNitrousOxide
    add ToxicSum ToxicSum TempReg
    l TempReg   SensorRoom RatioMethane
    add ToxicSum ToxicSum TempReg
    l TempReg   SensorRoom RatioWater
    add ToxicSum ToxicSum TempReg

    # 3. Lokale gevarencontrole
    sgt TempReg ToxicSum MAX_TOXIC_RATIO
    or HazardActive HazardActive TempReg

    # 4. Ventilatiestatus bepalen
    beqz HazardActive normal_balance

    # ALARM: Toevoer stoppen, compartiment actief purgen
    move SupOn 0
    move ExtOn 1
    j apply_room_vents

normal_balance:
    # NORMALE DRUKREGELING: Schakelen op basis van bandbreedte
    slt SupOn RoomPress PRESS_MIN
    sgt ExtOn RoomPress PRESS_MAX

apply_room_vents:
    # 5. Aansturing Active Vents
    s VentExtract Mode MODE_INWARD
    s VentExtract On ExtOn

    s VentSupply Mode MODE_OUTWARD
    s VentSupply On SupOn

    # 6. Exporteer lokale druk naar Logic Memory voor consoles
    s MemPressure Setting RoomPress

    j main
Uitleg & Afstelling: Kan op elke behuizing geladen worden om een compartiment via 2 Active Vents strak op druk te houden en live data naar Modular Consoles te sturen. In een sluisruimte waar je liever een lagere overgangsdruk aanhoudt, stel je PRESS_MIN en PRESS_MAX in op bijvoorbeeld 50.0 en 52.0.
